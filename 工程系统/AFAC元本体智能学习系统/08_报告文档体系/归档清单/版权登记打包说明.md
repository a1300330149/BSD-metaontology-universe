## 🔐 哈希校验方法

### SHA256哈希计算

### Windows PowerShell

```powershell
# 计算单个文件哈希
Get-FileHash -Algorithm SHA256 "文件路径"

# 批量计算目录下所有文件哈希
Get-ChildItem -Recurse | ForEach-Object { Get-FileHash -Algorithm SHA256 $_.FullName }
```

### Linux/Mac

```bash
# 计算单个文件哈希
sha256sum 文件路径

# 批量计算目录下所有文件哈希
find . -type f -exec sha256sum {} \; > hash_checksums.txt
```

```powershell
# 读取哈希文件
$hashes = Get-Content hash_checksums.txt

# 校验文件完整性
foreach ($hash in $hashes) {
    $parts = $hash -split '\s+'
    $file = $parts[2]
    $expected = $parts[0]
    $actual = (Get-FileHash -Algorithm SHA256 $file).Hash.ToLower()

    if ($actual -eq $expected) {
        Write-Host "$file: OK" -ForegroundColor Green
    } else {
        Write-Host "$file: FAILED" -ForegroundColor Red
    }
}
```

### 方法二：自动化打包脚本

```python
# scripts/package_for_submission.py
import os
import hashlib
import tarfile
from pathlib import Path

def calculate_sha256(file_path):
    """计算文件的SHA256哈希值"""
    sha256_hash = hashlib.sha256()
    with open(file_path, "rb") as f:
        for byte_block in iter(lambda: f.read(4096), b""):
            sha256_hash.update(byte_block)
    return sha256_hash.hexdigest()

def create_package():
    """创建打包文件"""
    package_dir = "Agent_v2_V3.0_Rev.04_Package"
    source_dir = "Agent_v2"

    # 创建打包目录
    os.makedirs(package_dir, exist_ok=True)

    # 复制所有文件
    for root, dirs, files in os.walk(source_dir):
        for file in files:
            src_path = os.path.join(root, file)
            rel_path = os.path.relpath(src_path, source_dir)
            dst_path = os.path.join(package_dir, rel_path)

            os.makedirs(os.path.dirname(dst_path), exist_ok=True)
            with open(src_path, 'rb') as src, open(dst_path, 'wb') as dst:
                dst.write(src.read())

    # 生成哈希校验文件
    hash_file = os.path.join(package_dir, "hash_checksums.txt")
    with open(hash_file, 'w') as f:
        for root, dirs, files in os.walk(package_dir):
            for file in files:
                if file == "hash_checksums.txt":
                    continue
                file_path = os.path.join(root, file)
                rel_path = os.path.relpath(file_path, package_dir)
                file_hash = calculate_sha256(file_path)
                f.write(f"{file_hash}  {rel_path}\n")

    # 打包压缩
    with tarfile.open("Agent_v2_V3.0_Rev.04_Package.tar.gz", "w:gz") as tar:
        tar.add(package_dir)

    print(f"打包完成：Agent_v2_V3.0_Rev.04_Package.tar.gz")

if __name__ == "__main__":
    create_package()
```
