import torch
import torch.nn as nn
import torch.nn.functional as F
from torch_geometric.nn import GCNConv, GATConv, TGNMemory, MessageNorm

# 一阶基础MLP
class MLP(nn.Module):
    def __init__(self, in_dim, hidden_dim, out_dim, num_layers=2):
        super().__init__()
        layers = []
        dims = [in_dim] + [hidden_dim]*(num_layers-1) + [out_dim]
        for i in range(len(dims)-1):
            layers.append(nn.Linear(dims[i], dims[i+1]))
            if i < len(dims)-2:
                layers.append(nn.ReLU())
                layers.append(nn.Dropout(0.2))
        self.net = nn.Sequential(*layers)
    def forward(self, x, edge_index=None):
        return self.net(x)

# 二阶GCN
class GCN(nn.Module):
    def __init__(self, in_dim, hidden_dim, out_dim, num_layers=2):
        super().__init__()
        self.convs = nn.ModuleList([GCNConv(in_dim, hidden_dim)])
        for _ in range(num_layers-2):
            self.convs.append(GCNConv(hidden_dim, hidden_dim))
        self.convs.append(GCNConv(hidden_dim))
        self.dropout = 0.2
    def forward(self, x, edge_index):
        for conv in self.convs[:-1]:
            x = F.relu(conv(x, edge_index))
            x = F.dropout(x, p=self.dropout, training=self.training)
        return self.convs[-1](x, edge_index)

# 三阶GAT
class GAT(nn.Module):
    def __init__(self, in_dim, hidden_dim, out_dim, num_heads=4):
        super().__init__()
        self.convs = nn.ModuleList([GATConv(in_dim, hidden_dim, heads=4)])
        self.convs.append(GATConv(hidden_dim*4, hidden_dim, heads=4))
        self.convs.append(GATConv(hidden_dim*4, out_dim, heads=1, concat=False))
        self.dropout = 0.2
    def forward(self, x, edge_index):
        for conv in self.convs[:-1]:
            x = F.elu(conv(x, edge_index))
            x = F.dropout(x, p=self.dropout)
        return self.convs[-1](x, edge_index)

# 四阶时序TGN
class TGN(nn.Module):
    def __init__(self, in_dim, hidden_dim, out_dim, num_nodes=10000):
        super().__init__()
        self.memory = TGNMemory(num_nodes, in_dim, 64, 32, MessageNorm(64), 'last')
        self.encoder = nn.Sequential(nn.Linear(64+in_dim, hidden_dim), nn.ReLU(), nn.Linear(hidden_dim, out_dim))
    def forward(self, x, edge_index, edge_time=None):
        if edge_time is not None:
            self.memory.update_state(edge_index[0], edge_index[1], edge_time, x[edge_index[1]])
        mem = self.memory.memory
        return self.encoder(torch.cat([x, mem[:x.size(0)]], dim=-1))
    def reset_memory(self):
        self.memory.reset_state() |

import torch

class BaseGraphAgent:
    def __init__(self, dataset_config, model_type="MLP"):
        self.dataset_config = dataset_config
        self.model_type = model_type
        self.model = None
        self.optimizer = None
        self.criterion = nn.CrossEntropyLoss()
        self.baseline_score = 0.0
        self.device = torch.device("cuda" if torch.cuda.is_available() else "cpu")

    def init_base_model(self, lr=0.001, wd=1e-4):
        d_in = self.dataset["feat_dim"]
        d_h = self.dataset["hidden_dim"]
        d_out = self.dataset["num_classes"]
        if self.model_type == "MLP":
            self.model = MLP(d_in, d_h, d_out)
        elif self.model_type == "GCN":
            self.model = GCN(d_in, d_h, d_out)
        elif self.model_type == "GAT":
            self.model = GAT(d_in, d_h, d_out)
        elif self.model_type == "TGN":
            self.model = TGN(d_in, d_h, d_out, self.dataset.get("num_nodes", 10000))
        self.model.to(self.device)
        self.optimizer = torch.optim.Adam(self.model.parameters(), lr=lr, weight_decay=wd)

    def train_one_epoch(self, data):
        data = data.to(self.device)
        self.model.train()
        self.optimizer.zero_grad()
        if self.model_type == "TGN" and hasattr(data, "edge_time"):
            out = self.model(data.x, data.edge_index, data.edge_time)
        else:
            out = self.model(data.x, data.edge_index)
        loss = self.criterion(out[data.train_mask], data.y[data.train_mask])
        loss.backward()
        self.optimizer.step()
        val_acc = evaluate(self.model, data, "val")

            self.baseline_score = val_acc
        return val_acc
